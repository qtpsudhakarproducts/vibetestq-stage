import { LoginPage } from './pages/LoginPage';

test('Login with valid credentials',
  async ({ page }) => {

  const login = new LoginPage(page);
  await login.goto();
  await login.loginWith(
    'Admin', 'admin123'
  );

  await expect(page)
    .toHaveURL('/dashboard');
});
